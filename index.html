-- ========================================== -- DATADEV DİNAMİK YÖNLENDİRİCİ CONFIG -- Oyun adında "FPS" geçen tüm oyunları kapsar. -- ========================================== local CONFIG = { { TargetKeyword = "FPS", -- Oyun adında "FPS" geçiyorsa eşleşir LoadUrl = "https://vorex-hack.vercel.app/single-tap-FPS.lua" } -- İstersen buraya ileride başka genel kurallar da ekleyebilirsin. } print("--- [DataDev] Genel FPS Oyun Algılama Sistemi Başlatıldı ---") -- Oyuncunun oynadığı oyunun adını güvenli bir şekilde çekme local MarketplaceService = game:GetService("MarketplaceService") local success, gameInfo = pcall(function() return MarketplaceService:GetProductInfo(game.PlaceId) end) local gameName = success and gameInfo.Name or "" print("Algılanan Oyun Adı: " .. (gameName ~= "" and gameName or "Bilinmiyor / Alınamadı")) print("Place ID: " .. tostring(game.PlaceId)) -- Eşleşme Kontrolü ve Yönlendirme local matched = false for _, data in ipairs(CONFIG) do -- Oyun adının içinde "FPS" kelimesi geçiyorsa (büyük/küçük harf duyarsız) if gameName ~= "" and string.find(string.lower(gameName), string.lower(data.TargetKeyword)) then matched = true print("[+] Eşleşme Bulundu! Eşleşen Kural: " .. data.TargetKeyword) print("[+] Yükleniyor -> " .. data.LoadUrl) -- Scripti çalıştırma local loadSuccess, loadErr = pcall(function() local remoteScript = game:HttpGet(data.LoadUrl) loadstring(remoteScript)() end) if not loadSuccess then warn("[-] Hedef script yüklenirken hata oluştu: " .. tostring(loadErr)) end break end end if not matched then print("[-] Bu oyunun adında 'FPS' kelimesi geçmiyor. İşlem sonlandırıldı.") end print("-------------------------------------------------------------")
